const express = require("express");
const sqlite3 = require("sqlite3").verbose();
const bcrypt = require("bcrypt");
const jwt = require("jsonwebtoken");

const app = express();
app.use(express.json());

const SECRET = "SECRETKEY";

// ================= DATABASE =================
const db = new sqlite3.Database("database.db");

db.serialize(() => {
  db.run(`CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    email TEXT UNIQUE,
    password TEXT,
    role TEXT
  )`);

  db.run(`CREATE TABLE IF NOT EXISTS products (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT,
    price INTEGER
  )`);

  bcrypt.hash("admin123", 10, (err, hash) => {
    db.run(
      "INSERT OR IGNORE INTO users (email,password,role) VALUES ('admin@site.com', ?, 'admin')",
      [hash]
    );
  });
});

// ================= HTML =================
const loginPage = `
<h2>Admin Login</h2>
<input id="email" placeholder="email"><br>
<input id="pass" type="password" placeholder="password"><br>
<button onclick="login()">Login</button>

<script>
function login(){
 fetch("/login",{
  method:"POST",
  headers:{"Content-Type":"application/json"},
  body:JSON.stringify({
    email:email.value,
    password:pass.value
  })
 }).then(r=>r.json()).then(d=>{
  if(d.token){
    localStorage.token=d.token;
    location="/admin";
  } else alert(d.error);
 });
}
</script>
`;

const adminPage = `
<h2>Admin Panel</h2>

<input id="name" placeholder="Product name">
<input id="price" placeholder="Price">
<button onclick="add()">Add Product</button>

<ul id="list"></ul>

<script>
function load(){
 fetch("/products").then(r=>r.json()).then(d=>{
  list.innerHTML="";
  d.forEach(p=>{
    list.innerHTML+=\`<li>\${p.name} - ₹\${p.price}
    <button onclick="del(\${p.id})">Delete</button></li>\`;
  })
 })
}

function add(){
 fetch("/admin/add",{
  method:"POST",
  headers:{
   "Content-Type":"application/json",
   "Authorization":localStorage.token
  },
  body:JSON.stringify({name:name.value,price:price.value})
 }).then(()=>load())
}

function del(id){
 fetch("/admin/delete/"+id,{
  method:"DELETE",
  headers:{ "Authorization":localStorage.token }
 }).then(()=>load())
}

load();
</script>
`;

// ================= AUTH =================
app.post("/login", (req, res) => {
  const { email, password } = req.body;

  db.get("SELECT * FROM users WHERE email=?", [email], (err, user) => {
    if (!user) return res.json({ error: "User not found" });

    bcrypt.compare(password, user.password, (err, ok) => {
      if (!ok) return res.json({ error: "Wrong password" });

      const token = jwt.sign({ role: user.role }, SECRET);
      res.json({ token });
    });
  });
});

function isAdmin(req, res, next) {
  const token = req.headers.authorization;
  if (!token) return res.sendStatus(403);

  const data = jwt.verify(token, SECRET);
  if (data.role !== "admin") return res.sendStatus(403);
  next();
}

// ================= ROUTES =================
app.get("/", (req, res) => res.send(loginPage));
app.get("/admin", (req, res) => res.send(adminPage));

app.post("/admin/add", isAdmin, (req, res) => {
  const { name, price } = req.body;
  db.run("INSERT INTO products (name,price) VALUES (?,?)", [name, price]);
  res.sendStatus(200);
});

app.delete("/admin/delete/:id", isAdmin, (req, res) => {
  db.run("DELETE FROM products WHERE id=?", [req.params.id]);
  res.sendStatus(200);
});

app.get("/products", (req, res) => {
  db.all("SELECT * FROM products", (err, rows) => res.json(rows));
});

// ================= START =================
app.listen(3000, () => {
  console.log("Running on http://localhost:3000");
});
